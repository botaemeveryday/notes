{{- /* utils/subject-groups.html Собирает секции content/posts в группы «предметов». Модель: одна папка = один вариант конспекта (преподаватель + автор). Варианты одного предмета связываются ключом `subject` во front matter _index.md. Если `subject` не задан — группой становится сама папка, то есть старые курсы продолжают работать без правок. Возвращает slice словарей: key — ключ группы (subject или slug папки, в нижнем регистре) title — заголовок предмета primary — Page основного варианта (accent / abbr / semester / description) variants — slice вариантов (см. ниже) teachers — уникальные имена преподавателей в порядке появления total — суммарное число лекций по всем вариантам semester — номер семестра (0 — «прочие материалы») Вариант: page, url, slug, teacher (строкой), teachers (slice), authors (slice имён), count Порядок вариантов внутри группы — по weight из _index.md, затем по фамилии преподавателя. Первый вариант считается основным. */ -}} {{- $byKey := dict -}} {{- with site.GetPage "section" "posts" -}} {{- range .Sections -}} {{- if and (not .Draft) (gt (len .RegularPages) 0) -}} {{- $slug := path.Base (strings.TrimSuffix "/" .RelPermalink) -}} {{- $key := lower (string (.Params.subject | default $slug)) -}} {{- $prev := index $byKey $key | default slice -}} {{- $byKey = merge $byKey (dict $key ($prev | append .)) -}} {{- end -}} {{- end -}} {{- end -}} {{- $groups := slice -}} {{- range $key, $sections := $byKey -}} {{- $variants := slice -}} {{- range $sections -}} {{- $sec := . -}} {{- $names := slice -}} {{- with $sec.Params.teachers -}} {{- range . -}} {{- with trim (string .) " " }}{{ $names = $names | append . }}{{ end -}} {{- end -}} {{- else -}} {{- with $sec.Params.teacher -}} {{- with trim (string .) " " }}{{ $names = $names | append . }}{{ end -}} {{- end -}} {{- end -}} {{- $authorNames := slice -}} {{- range partial "utils/authors.html" $sec.Params -}} {{- $authorNames = $authorNames | append .name -}} {{- end -}} {{- $w := int ($sec.Params.variantWeight | default $sec.Weight) -}} {{- $variants = $variants | append (dict "page" $sec "url" $sec.RelPermalink "slug" (path.Base (strings.TrimSuffix "/" $sec.RelPermalink)) "teacher" (delimit $names ", ") "teachers" $names "authors" $authorNames "count" (len $sec.RegularPages) "sortKey" (printf "%06d|%s|%s" $w (delimit $names ", ") $sec.Title) ) -}} {{- end -}} {{- $variants = sort $variants "sortKey" -}} {{- $primary := (index $variants 0).page -}} {{- $title := "" -}} {{- range $variants -}} {{- if not $title }}{{ with .page.Params.subjectTitle }}{{ $title = string . }}{{ end }}{{ end -}} {{- end -}} {{- if not $title }}{{ $title = $primary.Title }}{{ end -}} {{- $teachers := slice -}} {{- range $variants -}} {{- range .teachers -}} {{- if not (in $teachers .) }}{{ $teachers = $teachers | append . }}{{ end -}} {{- end -}} {{- end -}} {{- $total := 0 -}} {{- range $variants }}{{ $total = add $total .count }}{{ end -}} {{- $groups = $groups | append (dict "key" $key "title" $title "primary" $primary "variants" $variants "teachers" $teachers "total" $total "semester" (int ($primary.Params.semester | default 0)) "sortKey" (printf "%06d|%s" (int $primary.Weight) $title) ) -}} {{- end -}} {{- return (sort $groups "sortKey") -}}
